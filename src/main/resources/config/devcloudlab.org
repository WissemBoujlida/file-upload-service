# S3 API
server {
    listen 443 ssl http2;
    server_name s3.devcloudlab.org;
    ssl_certificate /etc/letsencrypt/live/s3.devcloudlab.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/s3.devcloudlab.org/privkey.pem; # managed by Certbot
    client_max_body_size 100M;
    proxy_buffering off;
    proxy_request_buffering off;
    proxy_send_timeout 120s;
    proxy_read_timeout 120s;

    location / {
        proxy_pass http://rustfs_api;
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
}

# CDN

    listen 443 ssl http2;
    server_name cdn.devcloudlab.org;
    ssl_certificate /etc/letsencrypt/live/cdn.devcloudlab.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/cdn.devcloudlab.org/privkey.pem;
    proxy_send_timeout 120s;

        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_http_version 1.1;
